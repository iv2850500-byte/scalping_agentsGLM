  330:             // бот мёртв, открытые позиции без exit-логики/трейлинга. Теперь
  331:             // после 3 быстрых попыток — вечный slow-retry каждые 600 сек
  332:             // (тяжёлый startup run_bot не спамим), transient-сбои биржи/WS
  333:             // само-восстанавливаются без ручного вмешательства.
  334:             const WATCHDOG_RETRY_SEC: u64 = 600;
  335:             for attempt in 1..=u32::MAX {
  336:                 let ws_rx = hub.subscribe();
  337:                 let shutdown_rx = shutdown_tx_bot.subscribe();
  338: 
  339:                 let inner = tokio::spawn(run_bot(
  340:                     bot_config.clone(),
  341:                     symbol.clone(),
  342:                     global_cfg.clone(),
  343:                     ws_rx,
  344:                     ts.clone(),
  345:                     shutdown_rx,
  346:                     filters.clone(),
  347:                     hub.clone(),
  348:                     global_risk_bot.clone(),
  349:                 ));
  350: 
  351:                 match inner.await {
  352:                     Ok(Ok(())) => return,
  353:                     Ok(Err(e)) => {
  354:                         tracing::error!(
  355:                             "[{}] ❌ бот упал (попытка {}/{}): {:?}",
  356:                             bot_name, attempt.min(MAX_ATTEMPTS), MAX_ATTEMPTS, e // AUDIT-P6: cap для slow-retry
  357:                         );
  358:                     }
  359:                     Err(join_err) => {
  360:                         tracing::error!(
  361:                             "[{}] 💥 задача бота завершена аварийно (паника: {}, попытка {}/{})",
  362:                             bot_name, join_err.is_panic(), attempt, MAX_ATTEMPTS
  363:                         );
  364:                     }
  365:                 }
  366: 
  367:                 metrics::ERRORS_TOTAL.with_label_values(&[&bot_name]).inc();
  368:                 metrics::BOTS_ACTIVE.with_label_values(&[&bot_name]).set(0);
  369: 
  370:                 if attempt == MAX_ATTEMPTS { // AUDIT-P6: сторож — return убран, цикл вечный
  371:                     tracing::error!(
  372:                         "[{}] 🚨 быстрые попытки ({}) исчерпаны — включаю СТОРОЖ: \
  373:                          retry каждые {} сек до успеха. ПОКА БОТ МЁРТВ — \
  374:                          ОТКРЫТЫЕ ПОЗИЦИИ БЕЗ ПРИСМОТРА, СВЕРЬ С БИРЖЕЙ",
  375:                         bot_name, MAX_ATTEMPTS, WATCHDOG_RETRY_SEC
  376:                     );
  377:                 }
  378:                 // AUDIT-P6: после исчерпания быстрых попыток — slow-retry.
  379:                 // PATCH P15: fast retries first, then 5 min, cap 30 min
  380:                 let backoff = std::time::Duration::from_secs(
  381:                     if attempt <= 2 { 5 * attempt as u64 }
  382:                     else if attempt <= 6 { 300 }
  383:                     else { 1800 }
  384:                 );
  385:                 tracing::warn!("[{}] перезапуск через {:?}...", bot_name, backoff);
  386:                 tokio::time::sleep(backoff).await;
  387:             }
  388:         });
  389:         handles.push(handle);
  390:     }
  391: 
  392:     // ============ GRACEFUL SHUTDOWN ============
  393:     #[cfg(unix)]
  394:     {
  395:         let mut sigterm =
  396:             tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate())?;
  397:         tokio::select! {
  398:             _ = tokio::signal::ctrl_c() => {},
  399:             _ = sigterm.recv() => {},
  400:         }
  401:     }
  402:     #[cfg(not(unix))]
  403:     {
  404:         tokio::signal::ctrl_c().await?;
  405:     }
  406:     tracing::warn!("🛑 Получен сигнал остановки — graceful shutdown (до 120 с: maker-выходы в плюс, потом маркет)...");
  407: 
  408:     let _ = shutdown_tx.send(());
  409: 
  410:     // ✅ МЯГКИЙ SHUTDOWN: дедлайн 120 с
  411:     let all = futures_util::future::join_all(handles);
  412:     match tokio::time::timeout(std::time::Duration::from_secs(120), all).await {
  413:         Ok(_) => tracing::info!("✅ все боты завершились корректно"),
  414:         Err(_) => tracing::error!(
  415:             "⏱ 120с истекли — часть ботов не завершилась, процесс завершается. СВЕРЬ ПОЗИЦИИ НА БИРЖЕ"
  416:         ),
  417:     }
  418: 
  419:     metrics_handle.abort();
  420:     tracing::info!("✅ Завершение завершено");
  421:     Ok(())
  422: }
