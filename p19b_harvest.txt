  495:                     }
  496:                     Some(Ok(Message::Text(text))) => {

  502:                         if is_bybit && (text.contains("\"op\":\"pong\"") || text.contains("\"op\": \"pong\"")) {
  503:                             last_pong_at = Instant::now();
  504:                             continue;
  505:                         }

  512:                             parse_okx_message(&text)

  514:                             parse_bybit_message(&text, &mut bt_cache)

  516:                             parse_message(&text)

  520:                             metrics::WS_MESSAGES_PARSED.with_label_values(&["global"]).inc();
  521: 
  522:                             match &m {
  523:                                 WebSocketMessage::Trade(_) => { health.trades.fetch_add(1, Ordering::Relaxed); }
  524:                                 WebSocketMessage::BookTicker(_) => { health.book_tickers.fetch_add(1, Ordering::Relaxed); }
  525:                                 WebSocketMessage::Kline(_) => { health.klines.fetch_add(1, Ordering::Relaxed); }
  526:                                 WebSocketMessage::OrderBook(_) => { health.order_books.fetch_add(1, Ordering::Relaxed); }
  527:                             }
  528: 
  529:                             if let WebSocketMessage::Trade(ref t) = m {
  530:                                 let local_ts_ms = chrono::Utc::now().timestamp_millis();
  531:                                 let raw_latency = (local_ts_ms - t.event_time).max(0) as u64;
  532:                                 if raw_latency < CLOCK_SKEW_THRESHOLD_MS {

  534:                                     health.latency_count.fetch_add(1, Ordering::Relaxed);

  539:                             }
  540: 
  541:                             let _ = tx.send(m);
  542:                         }
  543:                     }
  544:                     Some(Ok(Message::Close(frame))) => {
  545:                         tracing::warn!("Сервер закрыл: {:?}", frame);
  546:                         break;
  547:                     }
  548:                     Some(Err(e)) => {

  551:                     }

  555:                     }
  556:                     _ => {}
  557:                 }
  558:             }

  679: // v3.7b: Bybit v5 envelope parser {"topic":...,"type":...,"ts":...,"data":...}.
  680: // Handles exactly what we subscribe to: tickers.XAUUSDT (best bid/ask) and
  681: // publicTrade.XAUUSDT (trade tape). Everything else is silently ignored.
  682: fn parse_bybit_message(
  683:     text: &str,
  684:     bt_cache: &mut BybitTickerCache,
  685: ) -> Option<WebSocketMessage> {
  686:     let root: Value = serde_json::from_str(text).ok()?;
  687:     let topic = root.get("topic")?.as_str()?.to_string();
  688:     let root_ts = root.get("ts").and_then(|v| v.as_i64()).unwrap_or(0);
  689:     let data = root.get("data")?.clone();
  690: 
  691:     if topic.starts_with("tickers.") {
  692:         // data is an object; delta carries only the changed fields.
  693:         let get_f = |key: &str| -> Option<f64> {
  694:             data.get(key)
  695:                 .and_then(|v| v.as_str())
  696:                 .filter(|s| !s.is_empty())
  697:                 .and_then(|s| s.parse::<f64>().ok())
  698:         };
  699:         if let Some(v) = get_f("bid1Price") {
  700:             bt_cache.bid = v;
  701:         }
  702:         if let Some(v) = get_f("ask1Price") {
  703:             bt_cache.ask = v;
  704:         }
  705:         if let Some(v) = get_f("bid1Size") {
  706:             bt_cache.bid_qty = v;
  707:         }
  708:         if let Some(v) = get_f("ask1Size") {
  709:             bt_cache.ask_qty = v;
  710:         }
  711:         if bt_cache.bid <= 0.0 || bt_cache.ask <= 0.0 {
  712:             return None; // one side still unknown (wait for full snapshot)
  713:         }
  714:         let mut timestamp = data.get("timestamp").and_then(|v| v.as_i64()).unwrap_or(0);
  715:         if timestamp <= 0 {
  716:             timestamp = root_ts;
  717:         }
  718:         if timestamp <= 0 {
  719:             timestamp = chrono::Utc::now().timestamp_millis();
  720:         }
  721:         return Some(WebSocketMessage::BookTicker(BookTickerMessage {
  722:             symbol: data
  723:                 .get("symbol")
  724:                 .and_then(|v| v.as_str())
  725:                 .unwrap_or("UNKNOWN")
  726:                 .to_string(),
  727:             best_bid: bt_cache.bid,
  728:             best_ask: bt_cache.ask,
  729:             bid_qty: bt_cache.bid_qty,
  730:             ask_qty: bt_cache.ask_qty,
  731:             timestamp,
  732:         }));
  733:     }
  734: 
  735:     if topic.starts_with("publicTrade.") {
  736:         // data is an array: [{"T":ms,"s":"XAUUSDT","S":"Buy|Sell","v":..,"p":..}]
  737:         let arr = data.as_array()?;
  738:         let t = arr.last()?; // most recent trade in the batch
  739:         let price = t.get("p")?.as_str()?.parse::<f64>().ok()?;
  740:         let volume = t.get("v")?.as_str()?.parse::<f64>().ok()?;
  741:         let symbol = t.get("s")?.as_str()?.to_string();
  742:         let mut timestamp = t.get("T").and_then(|v| v.as_i64()).unwrap_or(0);
  743:         if timestamp <= 0 {
  744:             timestamp = root_ts;
  745:         }
  746:         let event_time = if root_ts > 0 { root_ts } else { timestamp };
  747:         // Bybit S="Buy" means aggressive buyer (taker buy) -> buyer is NOT maker.
  748:         let is_buyer_maker = t.get("S").and_then(|v| v.as_str()) == Some("Sell");
  749:         return Some(WebSocketMessage::Trade(TradeMessage {
  750:             symbol,
  751:             price,
  752:             volume,
  753:             timestamp,
  754:             event_time,
  755:             is_buyer_maker,
  756:         }));
  757:     }
  758: 
  759:     None
  760: }

  762: fn parse_message(text: &str) -> Option<WebSocketMessage> {
  763:     let root: Value = serde_json::from_str(text).ok()?;
  764: 
  765:     let (stream, data) = if let Some(s) = root.get("stream").and_then(|s| s.as_str()) {
  766:         (s.to_string(), root.get("data")?.clone())
  767:     } else {
  768:         (String::new(), root.clone())
  769:     };
  770: 
  771:     if stream.contains("@aggTrade") {
  772:         let symbol = data.get("s")?.as_str()?.to_string();
  773:         let price = data.get("p")?.as_str()?.parse::<f64>().ok()?;
  774:         let volume = data.get("q")?.as_str()?.parse::<f64>().ok()?;
  775:         let timestamp = data.get("T")?.as_i64().unwrap_or(0);
  776:         let event_time = data.get("E").and_then(|v| v.as_i64()).unwrap_or(timestamp);
  777:         let is_buyer_maker = data.get("m").and_then(|m| m.as_bool()).unwrap_or(false);
  778:         return Some(WebSocketMessage::Trade(TradeMessage {
  779:             symbol, price, volume, timestamp, event_time, is_buyer_maker,
  780:         }));
  781:     }
  782: 
  783:     if stream.contains("@bookTicker") {
  784:         let symbol = data.get("s")?.as_str()?.to_string();
  785:         let best_bid = data.get("b")?.as_str()?.parse::<f64>().ok()?;
  786:         let best_ask = data.get("a")?.as_str()?.parse::<f64>().ok()?;
  787:         let bid_qty = data.get("B").and_then(|v| v.as_str()).and_then(|s| s.parse().ok()).unwrap_or(0.0);
  788:         let ask_qty = data.get("A").and_then(|v| v.as_str()).and_then(|s| s.parse().ok()).unwrap_or(0.0);
  789:         let timestamp = data
  790:             .get("E")
  791:             .and_then(|v| v.as_i64())
  792:             .unwrap_or_else(|| chrono::Utc::now().timestamp_millis());
  793:         return Some(WebSocketMessage::BookTicker(BookTickerMessage {
  794:             symbol, best_bid, best_ask, bid_qty, ask_qty, timestamp,
  795:         }));
  796:     }
  797: 
  798:     if stream.contains("@depth") {
  799:         let symbol = data.get("s").and_then(|s| s.as_str()).unwrap_or("").to_string();
  800:         let bids = parse_levels(data.get("bids")?)?;
  801:         let asks = parse_levels(data.get("asks")?)?;
  802:         return Some(WebSocketMessage::OrderBook(OrderBookMessage { symbol, bids, asks }));
  803:     }
  804: 
  805:     if stream.contains("@kline") {
  806:         let k = data.get("k")?;
  807:         let symbol = k.get("s")?.as_str()?.to_string();
  808:         let open = k.get("o")?.as_str()?.parse::<f64>().ok()?;
  809:         let high = k.get("h")?.as_str()?.parse::<f64>().ok()?;
  810:         let low = k.get("l")?.as_str()?.parse::<f64>().ok()?;
  811:         let close = k.get("c")?.as_str()?.parse::<f64>().ok()?;
  812:         let volume = k.get("v")?.as_str()?.parse::<f64>().ok()?;
  813:         // ✅ taker buy base asset volume — k.V
  814:         let taker_buy_volume = k
  815:             .get("V")
  816:             .and_then(|v| v.as_str())
  817:             .and_then(|s| s.parse::<f64>().ok())
  818:             .unwrap_or(0.0);
  819:         // ✅ close_time — k.T
  820:         let timestamp = k.get("T").and_then(|v| v.as_i64()).unwrap_or(0);
  821:         let is_closed = k.get("x").and_then(|v| v.as_bool()).unwrap_or(false);
  822:         return Some(WebSocketMessage::Kline(KlineMessage {
  823:             symbol, open, high, low, close, volume, taker_buy_volume, timestamp, is_closed,
  824:             is_backfill: false,
  825:         }));
  826:     }
  827: 
  828:     None
  829: }

  831: fn parse_levels(v: &Value) -> Option<Vec<(f64, f64)>> {
  832:     Some(
  833:         v.as_array()?
  834:             .iter()
  835:             .filter_map(|level| {
  836:                 let arr = level.as_array()?;
  837:                 let price = arr.first()?.as_str()?.parse::<f64>().ok()?;
  838:                 let qty = arr.get(1)?.as_str()?.parse::<f64>().ok()?;
  839:                 Some((price, qty))
  840:             })
  841:             .collect(),
  842:     )
  843: }

  874:             .and_then(|s| s.parse::<i64>().ok())
  875:             .unwrap_or_else(|| chrono::Utc::now().timestamp_millis());
  876:         let symbol = okx_symbol_from_inst_id(t.get("instId").and_then(|v| v.as_str()).unwrap_or(""));

  882:             ask_qty,

  888:         let t = data.as_array()?.last()?;

  896:             price,
  897:             volume,

  899:             event_time: ts,
  900:             is_buyer_maker,
  901:         }));
  902:     }
  903: 
  904:     None
